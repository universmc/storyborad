# Développement d'une interface de configuration

Ce dépôt contient un fichier `configuration.json` servant à afficher une interface de configuration pour une page HTML.

## Fichier `configuration.json`

Le fichier `configuration.json` contient les données suivantes :
```json
{
  "title": "Configuration de l'application",
  "sections": [
    {
      "title": "Paramètres généraux",
      "fields": [
        {
          "name": "langue",
          "type": "select",
          "options": ["français", "anglais", "espagnol"]
        },
        {
          "name": "thème",
          "type": "select",
          "options": ["clair", "sombre"]
        },
        {
          "name": "tailleMinimalePolice",
          "type": "number",
          "min": 12,
          "max": 24
        }
      ]
    },
    {
      "title": "Paramètres avancés",
      "fields": [
        {
          "name": "activerCache",
          "type": "boolean"
        },
        {
          "name": "activerLog",
          "type": "boolean"
        },
        {
          "name": "nombreMaximumConnexions",

          "min": 1,
          "max": 10
        }

## Fichier `affichage.js`

Pour afficher l'interface de configuration dans une page HTML, il suffit d'utiliser le fichier JavaScript `affichage.js` suivant :
```javascript
// Récupération du fichier JSON
fetch('configuration.json')
  .then(response => response.json())
  .then(data => {
    // Création de la section "configuration"
    const configurationSection = document.createElement('section');
    configurationSection.id = 'configuration';

    // Création du titre de la section
    const configurationTitle = document.createElement('h2');
    configurationTitle.textContent = data.title;
    configurationSection.appendChild(configurationTitle);

    // Boucle sur les sections
    data.sections.forEach(section => {
      // Création de la section courante
      const currentSection = document.createElement('section');

      // Création du titre de la section courante
      const currentSectionTitle = document.createElement('h3');
      currentSectionTitle.textContent = section.title;
      currentSection.appendChild(currentSectionTitle);

      // Boucle sur les champs
      section.fields.forEach(field => {
        // Création du label
        const label = document.createElement('label');
        label.textContent = field.name;

        // Création du champ
        const input = document.createElement('input');
        input.type = field.type;
        input.name = field.name;

        switch (field.type) {
          case 'select':
            const select = document.createElement('select');
            field.options.forEach(option => {
              const optionElement = document.createElement('option');
              optionElement.textContent = option;
              select.appendChild(optionElement);
            });
            input.appendChild(select);
            break;
          case 'boolean':
            const trueCheckbox = document.createElement('input');
            trueCheckbox.type = 'radio';
            trueCheckbox.name = field.name;
            trueCheckbox.value = 'true';
            trueCheckbox.checked = true;
            const falseCheckbox = document.createElement('input');
            falseCheckbox.type = 'radio';
            falseCheckbox.name = field.name;
            falseCheckbox.value = 'false';
            input.appendChild(trueCheckbox);
            input.appendChild(falseCheckbox);
            break;
          case 'number':
            input.min = field.min;
            input.max = field.max;
            break;
          default:
            break;
        }

        label.appendChild(input);
        currentSection.appendChild(label);
      });

      configurationSection.appendChild(currentSection);
    });

    document.body.appendChild(configurationSection);
  });
```
Il ne reste plus qu'à inclure le fichier `affichage.js` dans la page HTML et le tour est joué !
