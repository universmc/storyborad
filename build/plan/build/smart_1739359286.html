SmartContrat Test Script


const { ethers } = require('hardhat');
const { expect } = require('chai');

describe('SmartContrat Test', function () {
  it('should check the output.sol file', async function () {
    const GeneratedContent = await ethers.getContractFactory('GeneratedContent');
    const generatedContent = await GeneratedContent.deploy();
    await generatedContent.deployed();

    expect(generatedContent.address).to.properAddress;
  });
});
